create table if not exists programmes_notifications (
  id bigint generated always as identity primary key,
  titre text not null,
  description text,
  chaine text,
  categorie text default 'match',
  heure_debut timestamptz not null,
  minutes_avant integer not null default 15,
  envoyee boolean not null default false,
  envoyee_at timestamptz,
  cree_par text,
  created_at timestamptz not null default now()
);
create index if not exists idx_prog_notif_envoi on programmes_notifications(envoyee, heure_debut);

create table if not exists device_notifications_recues (
  id bigint generated always as identity primary key,
  device_id text not null,
  notification_id bigint not null references programmes_notifications(id) on delete cascade,
  lue boolean not null default false,
  recue_at timestamptz not null default now()
);
create index if not exists idx_dnr_device on device_notifications_recues(device_id, lue);

alter table programmes_notifications enable row level security;
alter table device_notifications_recues enable row level security;
drop policy if exists "programmes_notifications_all" on programmes_notifications;
create policy "programmes_notifications_all" on programmes_notifications for all using (true) with check (true);
drop policy if exists "device_notifications_recues_all" on device_notifications_recues;
create policy "device_notifications_recues_all" on device_notifications_recues for all using (true) with check (true);
